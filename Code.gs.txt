const SHEET_NAME = 'Ventas';
const HEADERS = ['StoreCode', 'Date', 'Promoter', 'SP', 'Tab', 'WR', 'AUD', 'UltimaActualizacion'];
const CONFIG_SHEET_NAME = 'Config';

function doGet(e) {
  const action = e.parameter.action;
  const callback = e.parameter.callback;
  let result;

  if (action === 'getAll') {
    result = getAllSalesAsJson();
  } else if (action === 'getAnnouncement') {
    result = getAnnouncementData();
  } else if (action === 'save') {
    const storeCode = e.parameter.storeCode;
    const date = e.parameter.date;
    const promoter = e.parameter.promoter || 'P1';
    if (!storeCode || !date) {
      result = { ok: false, error: 'faltan storeCode o date' };
    } else {
      upsertRow(storeCode, date, promoter, {
        SP: Number(e.parameter.SP) || 0,
        Tab: Number(e.parameter.Tab) || 0,
        WR: Number(e.parameter.WR) || 0,
        AUD: Number(e.parameter.AUD) || 0
      });
      result = { ok: true };
    }
  } else {
    result = { error: 'accion no reconocida' };
  }

  return respond(result, callback);
}

/* Se deja doPost por compatibilidad, aunque la app ya no lo usa. */
function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const { storeCode, date, promoter } = body;
    if (!storeCode || !date) return respond({ ok: false, error: 'faltan storeCode o date' });
    upsertRow(storeCode, date, promoter || 'P1', {
      SP: Number(body.SP) || 0,
      Tab: Number(body.Tab) || 0,
      WR: Number(body.WR) || 0,
      AUD: Number(body.AUD) || 0
    });
    return respond({ ok: true });
  } catch (err) {
    return respond({ ok: false, error: String(err) });
  }
}

/* ---------------- comunicado editable ---------------- */
function getAnnouncementData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(CONFIG_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(CONFIG_SHEET_NAME);
    sheet.getRange('A1').setValue('Comunicado (edita la celda B1 — se refleja en la app al instante):');
    sheet.getRange('B1').setValue('¡Bienvenidos! A partir de hoy reportamos la venta diaria por acá.');
    sheet.getRange('A2').setValue('Imagen del comunicado (opcional): pega aquí la URL de una imagen publica (B2):');
    sheet.setColumnWidth(1, 420);
    sheet.setColumnWidth(2, 420);
  }
  const text = sheet.getRange('B1').getValue();
  const imageUrl = sheet.getRange('B2').getValue();
  return { text: text ? String(text) : '', imageUrl: imageUrl ? String(imageUrl) : '' };
}

/* ---------------- helpers internos ---------------- */

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(HEADERS);
    sheet.getRange('C2:C').setNumberFormat('@');
  } else {
    migrateSheetIfNeeded(sheet);
  }
  return sheet;
}

function migrateSheetIfNeeded(sheet) {
  const lastCol = Math.max(sheet.getLastColumn(), 1);
  const header = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  if (header.indexOf('Promoter') !== -1) return;

  sheet.insertColumnBefore(3);
  sheet.getRange(1, 3).setValue('Promoter');
  sheet.getRange('C2:C').setNumberFormat('@');

  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    const values = [];
    for (let i = 0; i < lastRow - 1; i++) values.push(['P1']);
    sheet.getRange(2, 3, lastRow - 1, 1).setValues(values);
  }
}

function normalizeDateValue(v) {
  if (v instanceof Date) {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  return String(v);
}

function upsertRow(storeCode, date, promoter, cats) {
  const sheet = getSheet();
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    const range = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
    for (let i = 0; i < range.length; i++) {
      if (range[i][0] === storeCode && normalizeDateValue(range[i][1]) === date && (range[i][2] || 'P1') === promoter) {
        const rowNum = i + 2;
        sheet.getRange(rowNum, 4, 1, 5).setValues([[cats.SP, cats.Tab, cats.WR, cats.AUD, new Date()]]);
        return;
      }
    }
  }
  sheet.appendRow([storeCode, date, promoter, cats.SP, cats.Tab, cats.WR, cats.AUD, new Date()]);
}

function getAllSalesAsJson() {
  const sheet = getSheet();
  const lastRow = sheet.getLastRow();
  const result = {};
  if (lastRow <= 1) return result;
  const data = sheet.getRange(2, 1, lastRow - 1, HEADERS.length).getValues();
  data.forEach(row => {
    const storeCode = row[0];
    if (!storeCode) return;
    const date = normalizeDateValue(row[1]);
    const promoter = row[2] || 'P1';
    if (!result[storeCode]) result[storeCode] = {};
    if (!result[storeCode][date]) result[storeCode][date] = {};
    result[storeCode][date][promoter] = { SP: row[3] || 0, Tab: row[4] || 0, WR: row[5] || 0, AUD: row[6] || 0 };
  });
  return result;
}

function respond(obj, callback) {
  const json = JSON.stringify(obj);
  if (callback) {
    return ContentService.createTextOutput(`${callback}(${json})`)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
